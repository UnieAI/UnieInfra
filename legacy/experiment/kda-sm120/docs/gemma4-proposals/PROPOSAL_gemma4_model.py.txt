"""Inference-only Gemma 4 model compatible with HuggingFace weights.

Text path only -- the vision tower is not implemented (see
``docs/gemma4-port-plan.md`` section 7).

Three things make Gemma 4 unlike every other model in this package, and all
three are easy to get subtly wrong:

1.  **Per-layer attention geometry.** ``sliding_attention`` layers are
    ``16 x 256``; ``full_attention`` layers are ``4 x 512``.  The full layers
    hold *fewer* KV bytes per token, but they are the unbounded ones, so they
    dominate long-context capacity.

2.  **``attention_k_eq_v``.** Full-attention layers have no ``v_proj``; V is
    derived from the K projection.  Crucially K and V do *not* end up equal:

        V = v_norm(k_proj(x))              -- v_norm has NO learned weight
        K = RoPE(k_norm(k_proj(x)))        -- k_norm HAS a learned weight

    so ``V = n`` and ``K = RoPE(n * w_k)`` where ``n = x / rms(x)``.  Aliasing
    the two in the cache is numerically wrong.  This implementation stores
    both, matching vLLM / SGLang / TensorRT-LLM.  The single-tensor cache is
    a follow-up that needs kernel support; the derivation above is what makes
    it possible.

3.  **Plain RMSNorm, not Gemma's ``(1 + w)``.**  Gemma 2 and 3 store
    ``weight - 1``; Gemma 4 does not.  ``GemmaRMSNorm`` in
    ``layers/layernorm.py`` implements the OLD convention and must NOT be used
    here.  Verified against the checkpoint: ``q_norm.weight ~ 1.02``,
    ``input_layernorm.weight`` mean ~4.9 -- these are direct multipliers.

Also easy to miss: every layer carries a scalar ``layer_scalar`` applied to
its output, and the final logits are soft-capped at 30.0.
"""

class Gemma4RotaryEmbedding(nn.Module):
    """Gemma 4 rotary embedding, covering both layer types.

    Gemma 4's ``"proportional"`` rope rotates only
    ``partial_rotary_factor * head_dim`` dimensions, but does it in a way that
    is *not* the usual "rotate the leading ``rotary_dim`` columns":

    * The frequency exponent divides by the **full** ``head_dim`` (512), not by
      the rotary dim (128).  The rotated pairs are a proportional slice off the
      low-frequency end of the full ladder.
    * The rotation is applied at full width with ``inv_freq`` zero-padded in
      the tail, so ``rotate_half`` pairs ``dim[i]`` with ``dim[i + head_dim/2]``
      (i.e. ``i + 256``), not ``dim[i + rotary_dim/2]`` (``i + 64``).  The
      zero-frequency tail is an identity rotation (cos=1, sin=0).

    Both details are load-bearing: getting either wrong yields plausible but
    degraded text rather than an error.  vLLM, SGLang and TensorRT-LLM each
    implement exactly this and each comment it as the trap.

    With ``rope_type="default"`` and ``partial_rotary_factor=1.0`` the same
    formula reduces to a standard neox rope, so sliding layers share the code
    path.
    """

    def __init__(
        self,
        head_dim: int,
        max_position_embeddings: int,
        base: float,
        partial_rotary_factor: float = 1.0,
        dtype: torch.dtype | None = None,
    ) -> None:
        super().__init__()
        self.head_dim = head_dim
        self.max_position_embeddings = max_position_embeddings
        self.base = base
        self.partial_rotary_factor = partial_rotary_factor

        # Number of rotated angle pairs, and the identity-rotation remainder.
        self.rope_angles = int(partial_rotary_factor * head_dim // 2)
        self.nope_angles = head_dim // 2 - self.rope_angles
        if self.rope_angles <= 0:
            raise ValueError(
                f"partial_rotary_factor={partial_rotary_factor} with "
                f"head_dim={head_dim} rotates no dimensions"
            )

        inv_freq = self._compute_inv_freq()
        cache = self._build_cache(inv_freq, dtype or torch.get_default_dtype())
        self.register_buffer("cos_sin_cache", cache, persistent=False)

    def _compute_inv_freq(self) -> torch.Tensor:
        # NOTE the denominator: head_dim, NOT 2 * rope_angles.
        exponents = (
            torch.arange(0, 2 * self.rope_angles, 2, dtype=torch.float32)
            / self.head_dim
        )
        inv_freq = 1.0 / (self.base**exponents)
        if self.nope_angles > 0:
            inv_freq = torch.cat(
                [inv_freq, torch.zeros(self.nope_angles, dtype=torch.float32)]
            )
        return inv_freq

    def _build_cache(self, inv_freq: torch.Tensor, dtype: torch.dtype):
        t = torch.arange(self.max_position_embeddings, dtype=torch.float32)
        freqs = torch.outer(t, inv_freq)
        return torch.cat([freqs.cos(), freqs.sin()], dim=-1).to(dtype)

    @staticmethod
    def _rotate_half(x: torch.Tensor) -> torch.Tensor:
        half = x.shape[-1] // 2
        return torch.cat((-x[..., half:], x[..., :half]), dim=-1)

    def forward(
        self,
        positions: torch.Tensor,
        query: torch.Tensor,
        key: torch.Tensor,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """Rotate ``query`` and ``key`` in place-equivalent fashion.

        Args:
            positions: ``[num_tokens]`` absolute positions.
            query: ``[num_tokens, num_heads * head_dim]``.
            key: ``[num_tokens, num_kv_heads * head_dim]``.

        Returns:
            The rotated ``(query, key)``, same shapes as the inputs.
        """
        cache = self.cos_sin_cache.index_select(0, positions)
        cos, sin = cache.chunk(2, dim=-1)
        # rotate_half pairs i with i + head_dim/2, so cos/sin are duplicated
        # across both halves.
        cos = torch.cat([cos, cos], dim=-1).unsqueeze(-2)
        sin = torch.cat([sin, sin], dim=-1).unsqueeze(-2)

        q = query.view(query.shape[0], -1, self.head_dim)
        k = key.view(key.shape[0], -1, self.head_dim)
        q = (q * cos) + (self._rotate_half(q) * sin)
        k = (k * cos) + (self._rotate_half(k) * sin)
        return q.flatten(-2, -1).to(query.dtype), k.flatten(-2, -1).to(key.dtype)

class Gemma4MLP(nn.Module):
    """Gated GELU MLP. Gemma 4 uses ``gelu_pytorch_tanh``, not SiLU."""

    def __init__(
        self,
        hidden_size: int,
        intermediate_size: int,
        hidden_activation: str,
        mapping: Mapping,
        quant_config: QuantizationConfig | None = None,
        reduce_results: bool = False,
        prefix: str = "",
    ) -> None:
        super().__init__()
        if hidden_activation not in ("gelu_pytorch_tanh", "gelu_tanh", "gelu"):
            raise ValueError(
                f"Gemma4 expects a tanh-approximated GELU activation, got "
                f"{hidden_activation!r}. Substituting SiLU here would be a "
                "silent accuracy regression."
            )
        self.gate_up_proj = MergedColumnParallelLinear(
            hidden_size,
            [intermediate_size] * 2,
            bias=False,
            quant_config=quant_config,
            tp_rank=mapping.dense.tp_rank,
            tp_size=mapping.dense.tp_size,
            tp_group=mapping.dense.tp_group,
            prefix=add_prefix("gate_up_proj", prefix),
        )
        self.down_proj = RowParallelLinear(
            intermediate_size,
            hidden_size,
            bias=False,
            quant_config=quant_config,
            reduce_results=reduce_results,
            tp_rank=mapping.dense.tp_rank,
            tp_size=mapping.dense.tp_size,
            tp_group=mapping.dense.tp_group,
            prefix=add_prefix("down_proj", prefix),
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        gate_up, _ = self.gate_up_proj(x)
        gate, up = gate_up.chunk(2, dim=-1)
        x, _ = self.down_proj(
            nn.functional.gelu(gate, approximate="tanh") * up
        )
        return x

class Gemma4Router(nn.Module):
    """Router preprocessing, applied to the router input only.

    ``weightless RMSNorm -> * hidden_size**-0.5 -> * learned per-dim scale ->
    bias-free projection to expert logits``.

    The projection is deliberately **replicated** rather than sharded: every
    rank must see the identical routing table before top-k dispatch, and a
    sharded (hence differently-rounded) projection would let ranks disagree on
    expert selection.
    """

    def __init__(self, config, prefix: str = "") -> None:
        super().__init__()
        self.hidden_size = config.hidden_size
        self.eps = config.rms_norm_eps
        self.root_size = self.hidden_size**-0.5
        self.scale = nn.Parameter(torch.ones(self.hidden_size))
        self.proj = ReplicatedLinear(
            self.hidden_size,
            config.num_experts,
            bias=False,
            quant_config=None,
            prefix=add_prefix("proj", prefix),
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x32 = x.float()
        norm = x32 * torch.rsqrt(x32.pow(2).mean(-1, keepdim=True) + self.eps)
        norm = norm * (self.scale.float() * self.root_size)
        logits, _ = self.proj(norm.to(x.dtype))
        return logits.float()


def gemma4_routing_function(
    gating_output: torch.Tensor,
    topk: int,
    per_expert_scale: torch.Tensor,
) -> tuple[torch.Tensor, torch.Tensor]:
    """Gemma 4 expert routing.

    Reference semantics: fp32 softmax over **all** experts, mask to top-k,
    renormalize, then multiply by a learned per-expert output scale.

    Because the weights are renormalized, ``softmax(all)[topk] / sum(...)`` is
    identically ``softmax(topk_logits)``, so the cheaper form below is exact
    rather than an approximation -- but only under renormalization.

    Args:
        gating_output: ``[num_tokens, num_experts]`` raw router logits.
        topk: Experts selected per token.
        per_expert_scale: ``[num_experts]`` learned output multipliers.

    Returns:
        ``(topk_weights fp32, topk_ids int32)``.
    """
    topk_logits, topk_ids = torch.topk(gating_output.float(), k=topk, dim=-1)
    topk_weights = torch.softmax(topk_logits, dim=-1)
    topk_weights = topk_weights * per_expert_scale[topk_ids].float()
    return topk_weights.to(torch.float32), topk_ids.to(torch.int32)

class Gemma4Attention(nn.Module):
    """Gemma 4 attention with per-layer geometry and optional K==V projection.

    ``scaling`` is 1.0, not ``head_dim**-0.5``: unlike Gemma 2/3, Gemma 4 has
    no ``query_pre_attn_scalar`` and folds the scale into the learned Q/K norm
    weights. All three reference implementations do this.
    """

    def __init__(
        self,
        config,
        mapping: Mapping,
        layer_id: int,
        quant_config: QuantizationConfig | None = None,
        prefix: str = "",
    ) -> None:
        super().__init__()
        self.config = config
        self.mapping = mapping
        self.layer_id = layer_id
        self.layer_type = g4.layer_types(config)[layer_id]
        self.is_full_attention = self.layer_type == g4.FULL_ATTENTION
        self.use_k_eq_v = g4.layer_uses_k_eq_v(config, layer_id)

        self.tp_size = mapping.attn.tp_size
        self.hidden_size = config.hidden_size
        self.total_num_heads = config.num_attention_heads
        self.total_num_kv_heads = g4.layer_num_kv_heads(config, layer_id)
        validate_attention_partition(
            self.total_num_heads, self.total_num_kv_heads, self.tp_size
        )
        self.num_heads = self.total_num_heads // self.tp_size
        self.num_kv_heads = max(1, self.total_num_kv_heads // self.tp_size)
        self.head_dim = g4.layer_head_dim(config, layer_id)
        self.q_size = self.num_heads * self.head_dim
        self.kv_size = self.num_kv_heads * self.head_dim
        # Gemma 4: no query_pre_attn_scalar; the Q/K norms carry the scaling.
        self.scaling = 1.0

        self.qkv_proj = QKVParallelLinear(
            self.hidden_size,
            self.head_dim,
            self.total_num_heads,
            self.total_num_kv_heads,
            bias=getattr(config, "attention_bias", False),
            quant_config=quant_config,
            tp_rank=mapping.attn.tp_rank,
            tp_size=mapping.attn.tp_size,
            tp_group=mapping.attn.tp_group,
            prefix=add_prefix("qkv_proj", prefix),
        )
        self.o_proj = RowParallelLinear(
            self.total_num_heads * self.head_dim,
            self.hidden_size,
            bias=getattr(config, "attention_bias", False),
            quant_config=quant_config,
            reduce_results=False,
            tp_rank=mapping.attn.tp_rank,
            tp_size=mapping.attn.tp_size,
            tp_group=mapping.attn.tp_group,
            prefix=add_prefix("o_proj", prefix),
        )

        # Q/K norms carry learned weights; the V norm does not (it is pure
        # normalization, and has no entry in the checkpoint).
        self.q_norm = RMSNorm(self.head_dim, eps=config.rms_norm_eps)
        self.k_norm = RMSNorm(self.head_dim, eps=config.rms_norm_eps)
        self.v_norm_eps = config.rms_norm_eps

        rope_params = g4.layer_rope_parameters(config, layer_id)
        self.rotary_emb = Gemma4RotaryEmbedding(
            head_dim=self.head_dim,
            max_position_embeddings=config.max_position_embeddings,
            base=float(rope_params["rope_theta"]),
            partial_rotary_factor=float(rope_params["partial_rotary_factor"]),
        )

        window = g4.layer_sliding_windows(config)[layer_id]
        self.attn = PagedAttention(
            self.num_heads,
            self.head_dim,
            self.scaling,
            num_kv_heads=self.num_kv_heads,
            layer_id=layer_id,
            group_id=self.layer_type,
            sliding_window_size=-1 if window is None else window,
        )

    def _norm_heads(self, x: torch.Tensor, weight: torch.Tensor | None):
        """RMSNorm over the last (head_dim) axis, optionally weighted."""
        x = x.view(x.shape[0], -1, self.head_dim).float()
        x = x * torch.rsqrt(x.pow(2).mean(-1, keepdim=True) + self.v_norm_eps)
        if weight is not None:
            x = x * weight.float()
        return x.flatten(-2, -1)

    def forward(
        self,
        positions: torch.Tensor,
        hidden_states: torch.Tensor,
        ctx: ForwardContext,
        out_cache_loc: torch.Tensor,
    ) -> torch.Tensor:
        qkv, _ = self.qkv_proj(hidden_states)
        q, k, v = qkv.split([self.q_size, self.kv_size, self.kv_size], dim=-1)

        dtype = hidden_states.dtype
        # ORDER MATTERS for k_eq_v layers: v_norm consumes the RAW k_proj
        # output, so V must be normalized before k_norm is applied. Computing
        # V first makes that independent of whether the ops are in-place.
        v = self._norm_heads(v, None).to(dtype)
        q = self._norm_heads(q, self.q_norm.weight).to(dtype)
        k = self._norm_heads(k, self.k_norm.weight).to(dtype)

        q, k = self.rotary_emb(positions, q, k)

        attn_output = self.attn(q, k, v, ctx, out_cache_loc)
        if attn_output.dim() == 3:
            attn_output = attn_output.reshape(attn_output.shape[0], -1)
        output, _ = self.o_proj(attn_output)
        return output

class Gemma4DecoderLayer(nn.Module):
    """Gemma 4 decoder layer.

    Residual structure (note the *four* norms -- Gemma 4 post-norms both the
    attention and the feed-forward output before adding the residual)::

        r = h
        h = attn(input_layernorm(r));  h = post_attention_layernorm(h) + r
        r = h
        h = mlp(pre_feedforward_layernorm(h))
        [ if MoE: h = post_ff_norm_1(h) + post_ff_norm_2(moe(pre_ff_norm_2(r))) ]
        h = post_feedforward_layernorm(h) + r
        h = h * layer_scalar

    ``layer_scalar`` is a per-layer learned scalar present in every Gemma 4
    checkpoint (0.089 at layer 0, 0.879 at layer 5 of the 31B). Dropping it
    does not crash; it silently corrupts the output distribution.

    In the MoE variant the dense MLP and the expert block run **in parallel**
    off the same residual and are summed. The router reads the *raw* residual,
    while the experts read ``pre_feedforward_layernorm_2`` of it.
    """

    def __init__(
        self,
        config,
        mapping: Mapping,
        layer_id: int = 0,
        quant_config: QuantizationConfig | None = None,
        prefix: str = "",
    ) -> None:
        super().__init__()
        self.config = config
        self.mapping = mapping
        self.layer_id = layer_id
        self.hidden_size = config.hidden_size

        self.self_attn = Gemma4Attention(
            config=config,
            mapping=mapping,
            layer_id=layer_id,
            quant_config=quant_config,
            prefix=add_prefix("self_attn", prefix),
        )
        self.mlp = Gemma4MLP(
            hidden_size=config.hidden_size,
            intermediate_size=config.intermediate_size,
            hidden_activation=config.hidden_activation,
            mapping=mapping,
            quant_config=quant_config,
            prefix=add_prefix("mlp", prefix),
        )

        self.input_layernorm = RMSNorm(config.hidden_size, eps=config.rms_norm_eps)
        self.post_attention_layernorm = RMSNorm(
            config.hidden_size, eps=config.rms_norm_eps
        )
        self.pre_feedforward_layernorm = RMSNorm(
            config.hidden_size, eps=config.rms_norm_eps
        )
        self.post_feedforward_layernorm = RMSNorm(
            config.hidden_size, eps=config.rms_norm_eps
        )

        self.enable_moe_block = bool(getattr(config, "enable_moe_block", False))
        if self.enable_moe_block:
            self.router = Gemma4Router(config, prefix=add_prefix("router", prefix))
            self.per_expert_scale = nn.Parameter(torch.ones(config.num_experts))
            self.pre_feedforward_layernorm_2 = RMSNorm(
                config.hidden_size, eps=config.rms_norm_eps
            )
            self.post_feedforward_layernorm_1 = RMSNorm(
                config.hidden_size, eps=config.rms_norm_eps
            )
            self.post_feedforward_layernorm_2 = RMSNorm(
                config.hidden_size, eps=config.rms_norm_eps
            )
            self.experts = _build_experts(config, mapping, quant_config, prefix)
        else:
            self.router = None
            self.experts = None

        # Per-layer output scalar, loaded from the checkpoint.
        self.register_buffer("layer_scalar", torch.ones(1), persistent=True)

    def forward(
        self,
        positions: torch.Tensor,
        hidden_states: torch.Tensor,
        ctx: ForwardContext,
        out_cache_loc: torch.Tensor,
        residual: torch.Tensor | None = None,
    ) -> tuple[torch.Tensor, None]:
        # Gemma 4 does not thread a fused residual between layers: each layer
        # re-reads its own input as the residual.
        residual = hidden_states
        hidden_states = self.input_layernorm(hidden_states)
        hidden_states = self.self_attn(
            positions, hidden_states, ctx, out_cache_loc
        )
        hidden_states = self.post_attention_layernorm(hidden_states)
        hidden_states = hidden_states + residual

        residual = hidden_states
        mlp_out = self.mlp(self.pre_feedforward_layernorm(hidden_states))

        if self.enable_moe_block:
            router_logits = self.router(residual)
            moe_in = self.pre_feedforward_layernorm_2(residual)
            moe_out = self._run_experts(moe_in, router_logits)
            mlp_out = self.post_feedforward_layernorm_1(
                mlp_out
            ) + self.post_feedforward_layernorm_2(moe_out)

        hidden_states = self.post_feedforward_layernorm(mlp_out) + residual
        hidden_states = hidden_states * self.layer_scalar
        return hidden_states, None

    def _run_experts(self, x: torch.Tensor, router_logits: torch.Tensor):
        topk_weights, topk_ids = gemma4_routing_function(
            router_logits,
            self.config.top_k_experts,
            self.per_expert_scale,
        )
        return self.experts(x, topk_weights, topk_ids)


def _build_experts(config, mapping, quant_config, prefix):
    """Construct the routed-expert block.

    Split out so the dense path never imports the MoE stack.

    NOTE: Gemma 4's experts use a **gated GELU** (``geglu``) activation with
    ``moe_intermediate_size`` (704 on the 26B-A4B), not the SiLU that most
    MoE models in this package use. The expert layer must be configured
    accordingly; substituting SiLU is a silent accuracy regression.
    """
    from tokenspeed.runtime.layers.moe.expert import MoELayer

    return MoELayer(
        top_k=config.top_k_experts,
        num_experts=config.num_experts,
        hidden_size=config.hidden_size,
        intermediate_size=config.moe_intermediate_size,
        quant_config=quant_config,
        layer_index=get_layer_id(prefix) or 0,
        prefix=add_prefix("experts", prefix),
        tp_rank=mapping.moe.tp_rank,
        tp_size=mapping.moe.tp_size,
        ep_rank=mapping.moe.ep_rank,
        ep_size=mapping.moe.ep_size,
    )

class Gemma4Model(nn.Module):
    """Gemma 4 text backbone."""

    def __init__(
        self,
        config,
        mapping: Mapping,
        quant_config: QuantizationConfig | None = None,
        prefix: str = "",
    ) -> None:
        super().__init__()
        self.config = config
        self.mapping = mapping
        self.vocab_size = config.vocab_size
        self.embed_tokens = VocabParallelEmbedding(
            config.vocab_size,
            config.hidden_size,
            quant_config=quant_config,
            tp_rank=mapping.attn.tp_rank,
            tp_size=mapping.attn.tp_size,
            tp_group=mapping.attn.tp_group,
        )
        self.layers = make_layers(
            config.num_hidden_layers,
            lambda idx, prefix: Gemma4DecoderLayer(
                config=config,
                mapping=mapping,
                layer_id=idx,
                quant_config=quant_config,
                prefix=prefix,
            ),
        )
        self.norm = RMSNorm(config.hidden_size, eps=config.rms_norm_eps)
        # Gemma scales the embedding by sqrt(hidden_size). This is a constant,
        # not a checkpoint tensor, and omitting it is catastrophic.
        self.normalizer = float(config.hidden_size**0.5)

    def get_input_embeddings(self, input_ids: torch.Tensor) -> torch.Tensor:
        return self.embed_tokens(input_ids) * self.normalizer

    def forward(
        self,
        input_ids: torch.Tensor,
        positions: torch.Tensor,
        ctx: ForwardContext,
        out_cache_loc: torch.Tensor,
        input_embeds: torch.Tensor | None = None,
    ) -> tuple[torch.Tensor, None]:
        if input_embeds is None:
            hidden_states = self.get_input_embeddings(input_ids)
        else:
            hidden_states = input_embeds

        residual = None
        for layer in self.layers:
            hidden_states, residual = layer(
                positions, hidden_states, ctx, out_cache_loc, residual
            )
        hidden_states = self.norm(hidden_states)
        return hidden_states, None


class Gemma4ForCausalLM(BaseCausalLM):
    """Gemma 4 causal LM (text only).

    ``final_logit_softcapping`` is applied by the shared ``LogitsProcessor``,
    which reads it off ``self.config`` -- so the config handed down here must
    be the *text* config, which is what :class:`Gemma4ForConditionalGeneration`
    passes.
    """

    model_cls = Gemma4Model

    bitsandbytes_stacked_params_mapping = {
        "q_proj": ("qkv_proj", 0),
        "k_proj": ("qkv_proj", 1),
        "v_proj": ("qkv_proj", 2),
        "gate_proj": ("gate_up_proj", 0),
        "up_proj": ("gate_up_proj", 1),
    }

    def __init__(
        self,
        config,
        mapping: Mapping,
        quant_config: QuantizationConfig | None = None,
    ) -> None:
        g4.validate(config)
        super().__init__(config=config, mapping=mapping, quant_config=quant_config)

    def get_input_embeddings(self, input_ids: torch.Tensor) -> torch.Tensor:
        return self.model.get_input_embeddings(input_ids)

    # -- weight loading -----------------------------------------------------

    def _iter_weights(self, weights):
        """Normalize checkpoint names and synthesize the missing ``v_proj``.

        Two transforms:

        * ``model.language_model.*`` -> ``model.*``. Gemma 4 checkpoints nest
          the text stack under the multimodal wrapper even when the vision
          tower is unused.
        * ``full_attention`` layers have no ``v_proj``; emit the ``k_proj``
          tensor a second time under the ``v_proj`` name so the fused QKV
          projection produces ``v_raw == k_raw``.

        The duplicate is yielded as the *same* tensor object rather than a
        clone -- the loader copies into the fused parameter, so a clone would
        only add transient host memory (~220 MB across the 31B's 10 full
        layers).
        """
        k_eq_v = g4.k_eq_v_layer_ids(self.config)
        for name, weight in weights:
            if name.startswith("model.language_model."):
                name = "model." + name[len("model.language_model.") :]
            elif name.startswith("language_model."):
                name = "model." + name[len("language_model.") :]

            if name.startswith("model.vision_tower") or name.startswith(
                "model.embed_vision"
            ):
                # Text-only port: the vision tower is not built.
                continue

            yield name, weight

            if ".self_attn.k_proj." in name:
                layer_id = get_layer_id(name)
                if layer_id is not None and layer_id in k_eq_v:
                    yield name.replace(".k_proj.", ".v_proj."), weight

    def load_weights(self, weights: Iterable[tuple[str, torch.Tensor]]):
        stacked_params_mapping = [
            ("qkv_proj", "q_proj", "q"),
            ("qkv_proj", "k_proj", "k"),
            ("qkv_proj", "v_proj", "v"),
            ("gate_up_proj", "gate_proj", 0),
            ("gate_up_proj", "up_proj", 1),
        ]
        params_dict = dict(self.named_parameters())
        buffers_dict = dict(self.named_buffers())

        for name, loaded_weight in self._iter_weights(weights):
            if "rotary_emb.inv_freq" in name:
                continue
            if self.config.tie_word_embeddings and "lm_head.weight" in name:
                continue

            # layer_scalar is a buffer, not a parameter.
            if name.endswith(".layer_scalar"):
                buf = buffers_dict.get(name)
                if buf is None:
                    continue
                buf.data.copy_(loaded_weight.to(buf.dtype).reshape(buf.shape))
                continue

            for param_name, weight_name, shard_id in stacked_params_mapping:
                if weight_name not in name:
                    continue
                mapped = name.replace(weight_name, param_name)
                if mapped.endswith(".bias") and mapped not in params_dict:
                    continue
                if mapped not in params_dict:
                    continue
                param = params_dict[mapped]
                param.weight_loader(param, loaded_weight, shard_id)
                break
            else:
                if name.endswith(".bias") and name not in params_dict:
                    continue
                if name not in params_dict:
                    # Unmapped MoE expert tensors are handled by the expert
                    # checkpoint loader; anything else is a genuine surprise.
                    if ".experts." in name or ".router." in name:
                        continue
                    raise KeyError(f"unexpected Gemma4 weight {name!r}")
                param = params_dict[name]
                weight_loader = getattr(param, "weight_loader", default_weight_loader)
                weight_loader(param, loaded_weight)

    def get_embed_and_head(self):
        return self.model.embed_tokens.weight, self.lm_head.weight

    def set_embed_and_head(self, embed, head):
        del self.model.embed_tokens.weight
        del self.lm_head.weight
        self.model.embed_tokens.weight = embed
        self.lm_head.weight = head
        torch.cuda.empty_cache()
        torch.cuda.synchronize()


class Gemma4ForConditionalGeneration(Gemma4ForCausalLM):
    """Entry point for ``Gemma4ForConditionalGeneration`` checkpoints.

    Text path only. The vision tower is not built and its weights are skipped
    at load time, so image inputs are not supported -- feeding them would
    silently drop the image rather than error, which is why the multimodal
    entry points are deliberately not implemented here.
    """

    def __init__(
        self,
        config,
        mapping: Mapping,
        quant_config: QuantizationConfig | None = None,
    ) -> None:
        text_config = g4.get_text_config(config)
        super().__init__(
            config=text_config, mapping=mapping, quant_config=quant_config
        )


EntryClass = [Gemma4ForCausalLM, Gemma4ForConditionalGeneration]
